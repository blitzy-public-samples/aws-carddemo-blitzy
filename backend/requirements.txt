# CardDemo Backend - Python dependency manifest (pip).
#
# Target runtime: Python 3.13.x (AAP 0.6.1; supported range 3.10-3.13, < 3.14).
# Install into a virtual environment, for example:
#   python3.13 -m venv .venv && source .venv/bin/activate
#   pip install -r requirements.txt
#
# All versions are pinned per Agent Action Plan 0.6.1 (validated against current
# stable releases, mid-2026). This manifest covers the FastAPI backend ONLY.
# `typer` is intentionally NOT listed here: it is a batch/ package dependency
# (batch/cli.py) per AAP 0.6.1, not a backend dependency.
#
# RUNTIME-ONLY set (QA finding M-22): this file lists ONLY the packages the
# running FastAPI service needs. The backend Dockerfile installs exactly this
# file, so the production image never carries test tooling. Development / test
# dependencies (pytest, pytest-asyncio, httpx) live in the companion
# `requirements-dev.txt`, which pulls this file in via `-r requirements.txt` and
# then adds the test set. Install for development with:
#   pip install -r requirements-dev.txt
# Both manifests mirror the pins in pyproject.toml (runtime `dependencies` vs the
# `dev` optional-dependencies group); keep all three synchronized.

# Web framework & server
fastapi==0.136.1
# starlette is pinned as a DIRECT dependency (QA finding M-15): app/main.py imports
# from `starlette.datastructures`, `starlette.middleware.base`, and
# `starlette.types` directly (the SecurityHeaders / CSRF / correlation-id
# middleware), so its version is part of this service's own contract and must not
# float with FastAPI's transitive `starlette>=0.46.0` range. 1.3.1 is the version
# FastAPI 0.136.1 resolves and the release whose `starlette.requests` hard-imports
# python_multipart (see the python-multipart pin below).
starlette==1.3.1
uvicorn[standard]==0.34.0
# python-multipart >= 0.0.32 (QA finding M-23): releases < 0.0.32 are vulnerable
# to a malformed-multipart denial-of-service. It is NOT removable even though this
# service declares no Form/File endpoints: Starlette 1.3.1 hard-imports
# `python_multipart` at `starlette.requests` module load, so the app cannot import
# without it. 0.0.32 is the current patched release.
python-multipart==0.0.32

# Data validation & configuration
pydantic==2.10.4
pydantic-settings==2.7.1
email-validator==2.2.0

# ORM & database
# SQLAlchemy's async engine requires greenlet; the [asyncio] extra pulls it in.
# greenlet is pinned explicitly for reproducible builds.
sqlalchemy[asyncio]==2.0.36
greenlet==3.5.3
alembic==1.14.0
asyncpg==0.30.0
psycopg2-binary==2.9.10

# Security / auth
passlib[bcrypt]==1.7.4
# bcrypt is pinned to 4.0.1: passlib 1.7.4 reads bcrypt.__about__.__version__,
# which bcrypt >= 4.1 removed, breaking password hashing. 4.0.1 is the last fully
# compatible release (supports the SEC-USR-PWD plaintext -> hash uplift).
bcrypt==4.0.1
# PyJWT >= 2.13.0 (QA finding M-23): earlier pins carry known advisories affecting
# the JWT decode/verify path used by session authentication; 2.13.0 is patched.
PyJWT==2.13.0

# Reporting
# PDF/HTML statement & report generation; CSV output uses the stdlib `csv` module.
# reportlab 4.5.1 (QA finding I-01): 4.2.5 emitted a Python 3.13 AST deprecation
# warning from rl_safe_eval; 4.5.1 is the latest 4.x release that clears it (a 5.x
# major bump is intentionally avoided to minimize API-change risk).
reportlab==4.5.1
Jinja2==3.1.6

# NOTE (QA finding M-22): test tooling (pytest, pytest-asyncio, httpx) is
# intentionally NOT listed here so the production backend image stays runtime-
# only. Those packages now live in `requirements-dev.txt`.
